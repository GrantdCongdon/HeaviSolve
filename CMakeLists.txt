cmake_minimum_required(VERSION 3.25)
project(HeaviSolve VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME license_inventory COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tools/license_scan.py)
set_tests_properties(license_inventory PROPERTIES LABELS foundation)
option(HS_ENABLE_MFEM_SPIKES "Build research-only MFEM diagnostics" OFF)
if(HS_ENABLE_MFEM_SPIKES)
  find_package(MFEM 4.8 CONFIG REQUIRED)
  foreach(spike poisson cavity)
    add_executable(mfem_${spike} verification/mfem_${spike}.cpp)
    target_link_libraries(mfem_${spike} PRIVATE mfem)
    target_compile_options(mfem_${spike} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME mfem_${spike} COMMAND mfem_${spike})
    set_tests_properties(mfem_${spike} PROPERTIES LABELS "m0;numerical" TIMEOUT 300)
  endforeach()
else()
  add_test(NAME m0_requires_mfem COMMAND ${CMAKE_COMMAND} -E false)
  set_tests_properties(m0_requires_mfem PROPERTIES LABELS m0)
endif()

add_test(NAME m0_acceptance_gate COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tools/m0_gate.py)
set_tests_properties(m0_acceptance_gate PROPERTIES LABELS "m0;acceptance")
add_test(NAME transitive_license_policy COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/verification/license_audit_test.py)
set_tests_properties(transitive_license_policy PROPERTIES LABELS foundation)
