cmake_minimum_required(VERSION 3.25)
project(HeaviSolveComplexCavity LANGUAGES CXX)
find_package(MFEM 4.10 CONFIG REQUIRED)
set(HS_PETSC_ROOT "" CACHE PATH "Installed shared complex PETSc prefix")
set(HS_SLEPC_ROOT "" CACHE PATH "Installed shared SLEPc prefix")
if(HS_PETSC_ROOT OR HS_SLEPC_ROOT)
  if(NOT HS_PETSC_ROOT OR NOT HS_SLEPC_ROOT)
    message(FATAL_ERROR "Set both HS_PETSC_ROOT and HS_SLEPC_ROOT")
  endif()
  find_path(HS_PETSC_INCLUDE petsc.h PATHS "${HS_PETSC_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  find_path(HS_SLEPC_INCLUDE slepceps.h PATHS "${HS_SLEPC_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(HS_PETSC_LIBRARY petsc PATHS "${HS_PETSC_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  find_library(HS_SLEPC_LIBRARY slepc PATHS "${HS_SLEPC_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  if(HS_PETSC_LIBRARY MATCHES "\\.a$" OR HS_SLEPC_LIBRARY MATCHES "\\.a$")
    message(FATAL_ERROR "Explicit prefix mode requires shared libraries; use pkg-config for static dependencies")
  endif()
  add_library(hs_slepc INTERFACE)
  target_include_directories(hs_slepc SYSTEM INTERFACE "${HS_SLEPC_INCLUDE}" "${HS_PETSC_INCLUDE}")
  target_link_libraries(hs_slepc INTERFACE "${HS_SLEPC_LIBRARY}" "${HS_PETSC_LIBRARY}")
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SLEPC REQUIRED IMPORTED_TARGET SLEPc)
  add_library(hs_slepc ALIAS PkgConfig::SLEPC)
endif()
add_executable(complex-cavity cavity.cpp)
target_compile_features(complex-cavity PRIVATE cxx_std_20)
target_compile_options(complex-cavity PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(complex-cavity PRIVATE mfem hs_slepc)
enable_testing()
add_test(NAME complex_cavity COMMAND complex-cavity)
set_tests_properties(complex_cavity PROPERTIES LABELS "m0;numerical" TIMEOUT 300)
